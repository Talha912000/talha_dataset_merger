let data1=null, data2=null, result=null;
function readFile(file){
  return new Promise((resolve,reject)=>{
    const reader=new FileReader();
    reader.onerror=()=>reject(new Error("Could not read "+file.name));
    reader.onload=e=>{
      try{
        if(file.name.toLowerCase().endsWith(".csv")){
          const parsed=Papa.parse(e.target.result,{header:true,skipEmptyLines:true,dynamicTyping:false});
          if(parsed.errors.some(x=>x.type==="Delimiter")) throw new Error("Could not parse CSV.");
          resolve(parsed.data.map(row=>cleanRow(row)));
        }else{
          const workbook=XLSX.read(e.target.result,{type:"array",raw:false});
          const sheet=workbook.Sheets[workbook.SheetNames[0]];
          resolve(XLSX.utils.sheet_to_json(sheet,{defval:"",raw:false}).map(row=>cleanRow(row)));
        }
      }catch(err){reject(err);}
    };
    if(file.name.toLowerCase().endsWith(".csv")) reader.readAsText(file);
    else reader.readAsArrayBuffer(file);
  });
}
function cleanRow(row){
  const out={};
  for(const [k,v] of Object.entries(row)) out[String(k).trim()]=v==null?"":String(v);
  return out;
}
function populateSelect(select,rows){
  select.innerHTML="";
  const cols=Object.keys(rows[0]||{});
  for(const col of cols){
    const opt=document.createElement("option");
    opt.value=col;opt.textContent=col;select.appendChild(opt);
  }
  const preferred=cols.find(c=>c.toLowerCase().replace(/[^a-z0-9]/g,"")==="cnic");
  if(preferred)select.value=preferred;
  select.disabled=cols.length===0;
}
async function loadFiles(){
  result=null;
  document.getElementById("resultCard").style.display="none";
  const f1=document.getElementById("file1").files[0];
  const f2=document.getElementById("file2").files[0];
  document.getElementById("name1").textContent=f1?f1.name:"No file selected";
  document.getElementById("name2").textContent=f2?f2.name:"No file selected";
  document.getElementById("mergeBtn").disabled=true;
  if(!f1||!f2){data1=null;data2=null;return}
  try{
    showStatus("Reading files in your browser…","success");
    [data1,data2]=await Promise.all([readFile(f1),readFile(f2)]);
    if(!data1.length||!data2.length)throw new Error("One or both files have no data rows.");
    populateSelect(document.getElementById("key1"),data1);
    populateSelect(document.getElementById("key2"),data2);
    document.getElementById("mergeBtn").disabled=false;
    showStatus(`Files ready.\nDataset 1: ${data1.length.toLocaleString()} rows\nDataset 2: ${data2.length.toLocaleString()} rows`,"success");
  }catch(e){data1=null;data2=null;showStatus(e.message,"error")}
}
document.getElementById("file1").addEventListener("change",loadFiles);
document.getElementById("file2").addEventListener("change",loadFiles);
function showStatus(msg,type){
  const s=document.getElementById("status");
  s.className="status "+type;s.textContent=msg;
}
function mergeData(){
  try{
    if(!data1||!data2)throw new Error("Upload both datasets first.");
    const k1=document.getElementById("key1").value;
    const k2=document.getElementById("key2").value;
    if(!k1||!k2)throw new Error("Select a matching column in each dataset.");
    const how=document.querySelector('input[name="how"]:checked').value;
    const rightMap=new Map();
    data2.forEach((row,i)=>{
      const key=String(row[k2]??"").trim();
      if(key==="")return;
      if(!rightMap.has(key))rightMap.set(key,[]);
      rightMap.get(key).push(i);
    });
    const matchedRight=new Set();
    const rightCols=Object.keys(data2[0]||{});
    const leftCols=Object.keys(data1[0]||{});
    const outRightName={};
    rightCols.forEach(c=>{
      if(c===k2 && k1===k2){outRightName[c]=null;return}
      outRightName[c]=leftCols.includes(c)?c+"_dataset2":c;
    });
    const outLeftName={};
    leftCols.forEach(c=>outLeftName[c]=rightCols.includes(c)&&!(c===k1&&c===k2)?c+"_dataset1":c);
    const output=[];
    function combine(a,b){
      const row={};
      for(const c of leftCols)row[outLeftName[c]]=a?a[c]:"";
      for(const c of rightCols){
        if(outRightName[c]!==null)row[outRightName[c]]=b?b[c]:"";
      }
      return row;
    }
    data1.forEach(a=>{
      const key=String(a[k1]??"").trim();
      const matches=key===""?[]:(rightMap.get(key)||[]);
      if(matches.length){
        matches.forEach(i=>{matchedRight.add(i);output.push(combine(a,data2[i]));});
      }else if(how==="left"||how==="outer"){
        output.push(combine(a,null));
      }
    });
    if(how==="right"||how==="outer"){
      data2.forEach((b,i)=>{
        if(!matchedRight.has(i))output.push(combine(null,b));
      });
    }
    result=output;
    document.getElementById("resultCard").style.display="block";
    document.getElementById("summary").textContent=
      `Merge complete: ${result.length.toLocaleString()} rows × ${Object.keys(result[0]||{}).length} columns. Join type: ${how.toUpperCase()}.`;
    showStatus("Merge completed successfully.","success");
    const rows=result.slice(0,100);
    if(!rows.length){document.getElementById("preview").innerHTML="<p>No rows matched the selected join.</p>";return}
    const cols=Object.keys(rows[0]);
    const table=document.createElement("table");
    const thead=document.createElement("thead"),tr=document.createElement("tr");
    cols.forEach(c=>{const th=document.createElement("th");th.textContent=c;tr.appendChild(th)});
    thead.appendChild(tr);table.appendChild(thead);
    const tbody=document.createElement("tbody");
    rows.forEach(row=>{
      const tr=document.createElement("tr");
      cols.forEach(c=>{const td=document.createElement("td");td.textContent=row[c]??"";tr.appendChild(td)});
      tbody.appendChild(tr);
    });
    table.appendChild(tbody);
    const preview=document.getElementById("preview");
    preview.replaceChildren(table);
    if(result.length>100){const note=document.createElement("p");note.className="muted";note.textContent="Showing first 100 rows only. Download the file to get all records.";preview.appendChild(note)}
  }catch(e){showStatus(e.message,"error")}
}
function downloadData(){
  if(!result)return;
  const format=document.getElementById("format").value;
  let filename=document.getElementById("outname").value.trim()||"merged_dataset";
  filename=filename.replace(/[\\/:*?"<>|]/g,"_").replace(/\.(csv|xlsx)$/i,"");
  if(format==="csv"){
    const csv=Papa.unparse(result);
    const blob=new Blob(["\ufeff"+csv],{type:"text/csv;charset=utf-8"});
    saveBlob(blob,filename+".csv");
  }else{
    const wb=XLSX.utils.book_new();
    const ws=XLSX.utils.json_to_sheet(result);
    XLSX.utils.book_append_sheet(wb,ws,"Merged Data");
    XLSX.writeFile(wb,filename+".xlsx");
  }
}
function saveBlob(blob,name){
  const url=URL.createObjectURL(blob);
  const a=document.createElement("a");a.href=url;a.download=name;
  document.body.appendChild(a);a.click();a.remove();
  setTimeout(()=>URL.revokeObjectURL(url),1000);
}
function resetApp(){
  document.getElementById("file1").value="";
  document.getElementById("file2").value="";
  document.getElementById("name1").textContent="No file selected";
  document.getElementById("name2").textContent="No file selected";
  for(const id of ["key1","key2"]){
    const s=document.getElementById(id);s.innerHTML="<option>Upload files first</option>";s.disabled=true;
  }
  document.getElementById("mergeBtn").disabled=true;
  document.getElementById("resultCard").style.display="none";
  document.getElementById("status").className="status";
  data1=null;data2=null;result=null;
}
